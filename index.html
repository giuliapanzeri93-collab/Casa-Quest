-- CASA QUEST — MOTORE CHECK QUEST -- Una Check Quest è un controllo, non una faccenda obbligatoria. -- Se tutto è a posto: registriamo il controllo e assegniamo 0. -- Se c'era qualcosa da fare: viene registrato l'intervento e parte -- una normale richiesta di approvazione con il bonus della Check Quest. ALTER TABLE public.quest_completions ADD COLUMN IF NOT EXISTS check_action_taken boolean; ALTER TABLE public.quest_completions ADD COLUMN IF NOT EXISTS check_checked_at timestamptz; CREATE INDEX IF NOT EXISTS idx_quest_completions_check_history ON public.quest_completions (quest_id, player_id, completed_at) WHERE check_action_taken IS NOT NULL; CREATE OR REPLACE FUNCTION public.perform_check_quest( p_quest_id uuid, p_action_taken boolean, p_time_minutes integer DEFAULT NULL ) RETURNS public.quest_completions LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_user_id uuid; v_household_id uuid; v_quest public.quests; v_completion public.quest_completions; v_xp integer; v_coins integer; BEGIN v_user_id := auth.uid(); SELECT household_id INTO v_household_id FROM public.players WHERE id = v_user_id; IF v_household_id IS NULL THEN RAISE EXCEPTION 'Utente non collegato a una casa.'; END IF; SELECT * INTO v_quest FROM public.quests WHERE id = p_quest_id AND household_id = v_household_id AND active = true AND quest_type = 'check'; IF NOT FOUND THEN RAISE EXCEPTION 'Check Quest non trovata.'; END IF; IF p_time_minutes IS NOT NULL AND (p_time_minutes < 1 OR p_time_minutes > 480) THEN RAISE EXCEPTION 'Il tempo deve essere tra 1 e 480 minuti.'; END IF; -- Se non c'era nulla da fare, nessuna ricompensa. IF p_action_taken = false THEN v_xp := 0; v_coins := 0; ELSE -- Il bonus della Check Quest è la ricompensa per aver trovato -- e risolto effettivamente qualcosa. v_xp := v_quest.check_bonus_xp; v_coins := v_quest.check_bonus_coins; END IF; INSERT INTO public.quest_completions ( quest_id, player_id, completed_at, submitted_at, time_minutes, xp_awarded, coins_awarded, status, attempt_number, check_action_taken, check_checked_at ) VALUES ( p_quest_id, v_user_id, now(), now(), p_time_minutes, v_xp, v_coins, CASE WHEN p_action_taken THEN 'pending' ELSE 'approved' END, 1, p_action_taken, now() ) RETURNING * INTO v_completion; RETURN v_completion; END; $$; GRANT EXECUTE ON FUNCTION public.perform_check_quest( uuid, boolean, integer ) TO authenticated; -- Controllo finale SELECT column_name, data_type, column_default FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'quest_completions' AND column_name IN ( 'check_action_taken', 'check_checked_at' ) ORDER BY column_name;
